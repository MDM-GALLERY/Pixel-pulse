GroceryDataActor.swift
import SwiftData
@ModelActor
public actor GroceryDataActor {
public func fetchAllGroceryItems() async throws -> [GroceryItemDTO] {
let descriptor = FetchDescriptor<GroceryItem>(
sortBy: [SortDescriptor(\\.name)]
)
let items = try modelContext.fetch(descriptor)
return items.map { GroceryItemDTO(from: $0) }
}
}
